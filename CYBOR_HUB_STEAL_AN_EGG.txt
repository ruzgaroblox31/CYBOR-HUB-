local function chooseInputMode()
    local p = CG:FindFirstChild("CYBOR_ModeWarning") if p then p:Destroy() end
    local g = IN("ScreenGui", CG)
    g.Name, g.IgnoreGuiInset, g.ResetOnSpawn, g.DisplayOrder = "CYBOR_ModeWarning", true, false, 10000
    
    local f = IN("Frame", g)
    f.AnchorPoint, f.Position, f.Size = Vector2.new(0.5, 0.5), U2(0.5, 0, 0.5, 0), U2(0.88, 0, 0, 245)
    f.BackgroundColor3, f.BackgroundTransparency = C3(12, 12, 18), 0.23
    IN("UISizeConstraint", f).MaxSize = Vector2.new(485, 245)
    IN("UICorner", f).CornerRadius = UDim.new(0, 15)
    IN("UIStroke", f).Color, IN("UIStroke", f).Thickness = C3(255, 30, 60), 2
    
    local function label(t, y, h, s, c)
        local lb = IN("TextLabel", f)
        lb.Position, lb.Size, lb.BackgroundTransparency = U2(0, 12, 0, y), U2(1, -24, 0, h), 1
        lb.Text, lb.TextColor3, lb.TextSize, lb.Font, lb.TextWrapped = t, c, s, Enum.Font.GothamBold, true
        return lb
    end
    label("CYBOR HUB  |  INPUT MODE", 12, 34, 21, C3(255, 45, 70))
    label("PC Mode enables WASD, Jump, Shiftlock, 1-5 inventory slots and mouse controls.", 50, 102, 14, C3(240, 240, 245))
    
    local function btn(txt, pos, col)
        local bt = IN("TextButton", f)
        bt.Position, bt.Size = pos, U2(0.47, 0, 0, 49)
        bt.BackgroundColor3, bt.BackgroundTransparency, bt.TextColor3 = col, 0.16, C3(255, 255, 255)
        bt.Text, bt.TextScaled, bt.Font = txt, true, Enum.Font.GothamBold
        IN("UICorner", bt).CornerRadius = UDim.new(0, 9)
        return bt
    end
    
    local pc = btn("Continue (PC Mode)", U2(0.02, 0, 1, -67), C3(185, 28, 52))
    local mo = btn("Continue Mobile Mode (⚠)", U2(0.51, 0, 1, -67), C3(58, 58, 65))
    
    local d = IN("BindableEvent")
    pc.MouseButton1Click:Connect(function() d:Fire(true) end)
    mo.MouseButton1Click:Connect(function() d:Fire(false) end)
    local res = d.Event:Wait()
    d:Destroy() g:Destroy()
    return res
end

-- Sanal Klavye ve Mouse Motoru
local keyMap = {
    W=Enum.KeyCode.W, A=Enum.KeyCode.A, S=Enum.KeyCode.S, D=Enum.KeyCode.D,
    Space=Enum.KeyCode.Space, Shift=Enum.KeyCode.LeftShift, Ctrl=Enum.KeyCode.LeftControl,
    ["1"]=Enum.KeyCode.One, ["2"]=Enum.KeyCode.Two, ["3"]=Enum.KeyCode.Three, ["4"]=Enum.KeyCode.Four, ["5"]=Enum.KeyCode.Five
}
local activeKeys = {}

local function sendVirtualKey(kName, isDown)
    local kc = keyMap[kName] or Enum.KeyCode[kName]
    if not kc or activeKeys[kName] == isDown then return false end
    activeKeys[kName] = isDown
    local ok = false
    if VIM then ok = pcall(function() VIM:SendKeyEvent(isDown, kc, false, game) end) end
    return ok
end

local function sendVirtualClick(x, y)
    -- Topbar boşluğunu (GuiInset) hesapla, böylece tıklama tam parmağın altına denk gelir
    y = y + game:GetService("GuiService"):GetGuiInset().Y
    
    -- Ekranda kırmızı tıklama efekti oluştur
    local pcGui = CG:FindFirstChild("CYBOR_PC_Controls")
    if pcGui then
        local dot = pcGui:FindFirstChild("TouchClickDot")
        if dot then
            dot.Position = U2(0, x, 0, y)
            dot.Visible = true
            task.delay(0.18, function() if dot then dot.Visible = false end end)
        end
    end
    
    local clicked = false
    if VIM then
        clicked = pcall(function()
            VIM:SendMouseMoveEvent(x, y, game)
            task.wait(0.01)
            VIM:SendMouseButtonEvent(x, y, 0, true, game, 1)
            task.wait(0.02)
            VIM:SendMouseButtonEvent(x, y, 0, false, game, 1)
        end)
    elseif type(mouse1click) == "function" then
        clicked = pcall(function()
            if type(mousemoveabs) == "function" then mousemoveabs(x, y) end
            mouse1click()
        end)
    end
    
    local char = LP.Character
    local tool = char and char:FindFirstChildOfClass("Tool")
    if tool then pcall(function() tool:Activate() end) clicked = true end
    return clicked
end

-- PC Ekran Düzeni ve Dokunmatik Arayüz
local function createPCControls(sTab)
    if not PC_MODE then
        sTab:Section({Title="Mobile Mode"})
        sTab:Paragraph({Title="PC Controls Disabled", Desc="Shoot Murder and Ultra Auto Shoot are locked."})
        return
    end
    
    local Http = game:GetService("HttpService")
    local lFile = "CYBOR_PC_LAYOUT.json"
    local old = CG:FindFirstChild("CYBOR_PC_Controls")
    if old then old:Destroy() end
    
    local g = IN("ScreenGui", CG)
    g.Name, g.IgnoreGuiInset, g.ResetOnSpawn, g.DisplayOrder, g.ZIndexBehavior = "CYBOR_PC_Controls", true, false, 95, Enum.ZIndexBehavior.Sibling
    local cols = {default = C3(37,37,46), edit = C3(17,80,240), active = C3(185,28,52)}
    local editMode, mouseClickMode = false, true
    local controls, defaults, moveTouch, moveValue = {}, {}, nil, Vector2.zero
    local editTouch, editControl, editAction, editStartPos, editStartValue = nil, nil, nil, nil, nil
    local pendingMouseTouches = {}
    
    local function inRect(pos, el)
        local o, e = el.AbsolutePosition, el.AbsoluteSize
        return pos.X >= o.X and pos.X <= o.X + e.X and pos.Y >= o.Y and pos.Y <= o.Y + e.Y
    end
    
    local function addControl(name, cap, pos, sz, circ)
        local c = IN("Frame", g)
        c.Name, c.Size, c.Position = name, sz, pos
        c.BackgroundColor3, c.BackgroundTransparency = cols.default, circ and 0.27 or 0.55
        c.Active, c.ZIndex = true, 2
        IN("UICorner", c).CornerRadius = circ and UDim.new(1, 0) or UDim.new(0, 9)
        
        local st = IN("UIStroke", c)
        st.Color, st.Thickness = C3(190, 190, 202), 1.4
        
        local txt = IN("TextLabel", c)
        txt.Name, txt.BackgroundTransparency, txt.Size = "Caption", 1, U2(1,0,1,0)
        txt.Text, txt.TextColor3, txt.TextScaled, txt.Font, txt.ZIndex = cap, C3(255,255,255), true, Enum.Font.GothamBold, 3
        IN("UIPadding", txt).PaddingLeft, IN("UIPadding", txt).PaddingRight = UDim.new(0,5), UDim.new(0,5)
        
        local h = IN("TextButton", c)
        h.Name, h.Size, h.Position = "ResizeHandle", U2(0,19,0,19), U2(1,-19,1,-19)
        h.Text, h.TextScaled, h.TextColor3, h.BackgroundColor3 = "↘", true, C3(255,255,255), C3(40,125,255)
        h.ZIndex, h.Visible = 9, false
        IN("UICorner", h).CornerRadius = UDim.new(0, 4)
        
        controls[name] = {frame = c, stroke = st, resize = h, circular = circ, label = txt}
        defaults[name] = {p = pos, s = sz}
        return c
    end
    
    local stick = addControl("Joystick", "", U2(0, 23, 1, -202), U2(0, 165, 0, 165), true)
    local knob = IN("Frame", stick)
    knob.Name, knob.AnchorPoint, knob.Position, knob.Size = "Thumb", Vector2.new(0.5, 0.5), U2(0.5, 0, 0.5, 0), U2(0.29, 0, 0.29, 0)
    knob.BackgroundColor3, knob.BackgroundTransparency, knob.Active, knob.ZIndex = C3(255, 255, 255), 0.2, false, 4
    IN("UICorner", knob).CornerRadius = UDim.new(1, 0)
    
    local keyLabels = {}
    local function markKey(t, x, y)
        local l = IN("TextLabel", stick)
        l.Name, l.BackgroundTransparency, l.Size, l.Position = "Direction"..t, 1, U2(0.22,0,0.22,0), U2(x,0,y,0)
        l.Text, l.TextColor3, l.Font, l.TextScaled, l.Active, l.ZIndex = t, C3(250,250,255), Enum.Font.GothamBold, true, false, 3
        keyLabels[t] = l
    end
    markKey("W", 0.39, 0.06) markKey("A", 0.05, 0.4) markKey("D", 0.75, 0.4) markKey("S", 0.39, 0.75)
    
    addControl("Shift", "SHIFT", U2(0, 69, 1, -266), U2(0, 78, 0, 44), false)
    addControl("Ctrl", "CTRL", U2(0, 69, 1, -315), U2(0, 78, 0, 44), false)
    
    local slots = {}
    for i=1,5 do slots[i] = addControl("Slot"..i, tostring(i), U2(0.5, -150 + (i-1)*62, 1, -83), U2(0, 56, 0, 56), false) end
    
    addControl("Jump", "JUMP", U2(1, -126, 1, -210), U2(0, 58, 0, 58), true)
    addControl("Mouse", "LMB", U2(1, -126, 1, -143), U2(0, 58, 0, 58), false)
    addControl("WheelUp", "↑", U2(1, -57, 1, -191), U2(0, 42, 0, 43), false)
    addControl("WheelDown", "↓", U2(1, -57, 1, -141), U2(0, 42, 0, 43), false)
    
    local reticle = IN("TextLabel", g)
    reticle.Name, reticle.AnchorPoint, reticle.Position, reticle.Size = "LockedMouseCursor", Vector2.new(0.5, 0.5), U2(0.5,0,0.5,0), U2(0,30,0,30)
    reticle.BackgroundTransparency, reticle.Text, reticle.TextSize, reticle.Font = 1, "+", 30, Enum.Font.GothamBold
    reticle.TextColor3, reticle.TextStrokeTransparency, reticle.Active, reticle.ZIndex, reticle.Visible = C3(255,50,70), 0.2, false, 10, false
    
    local touchDot = IN("Frame", g)
    touchDot.Name, touchDot.AnchorPoint, touchDot.Size = "TouchClickDot", Vector2.new(0.5, 0.5), U2(0,22,0,22)
    touchDot.BackgroundColor3, touchDot.BackgroundTransparency, touchDot.Visible, touchDot.ZIndex = C3(255,35,65), 0.2, false, 20
    IN("UICorner", touchDot).CornerRadius = UDim.new(1, 0)

    local function updateMoveKeys(x, y)
        local w, s_key, a, d = y > 0.25, y < -0.25, x < -0.25, x > 0.25
        sendVirtualKey("W", w) sendVirtualKey("S", s_key) sendVirtualKey("A", a) sendVirtualKey("D", d)
        
        if keyLabels.W then keyLabels.W.TextColor3 = w and C3(255,45,70) or C3(250,250,255) end
        if keyLabels.S then keyLabels.S.TextColor3 = s_key and C3(255,45,70) or C3(250,250,255) end
        if keyLabels.A then keyLabels.A.TextColor3 = a and C3(255,45,70) or C3(250,250,255) end
        if keyLabels.D then keyLabels.D.TextColor3 = d and C3(255,45,70) or C3(250,250,255) end
    end
    
    local function setEditing(val)
        updateMoveKeys(0,0) editMode = val moveTouch, moveValue = nil, Vector2.zero
        editTouch, editControl, editAction = nil, nil, nil knob.Position = U2(0.5,0,0.5,0)
        for id, item in pairs(controls) do
            if id == "Shift" and cyRuntime.CYBOR_SHIFTLOCK and not val then
                item.frame.BackgroundColor3 = cols.active
            else
                item.frame.BackgroundColor3 = val and cols.edit or cols.default
            end
            item.stroke.Color = val and C3(75,170,255) or C3(190,190,202)
            item.resize.Visible = val
        end
    end
    
    local function restoreDefaults()
        for id, def in pairs(defaults) do
            controls[id].frame.Position = def.p
            controls[id].frame.Size = def.s
        end
    end
    
    local function loadLayout()
        if type(readfile) ~= "function" or type(isfile) ~= "function" or not isfile(lFile) then return end
        local ok, data = pcall(function() return Http:JSONDecode(readfile(lFile)) end)
        if ok and type(data) == "table" then
            for id, v in pairs(data) do
                if controls[id] and type(v) == "table" and #v == 8 then
                    controls[id].frame.Position = U2(v[1], v[2], v[3], v[4])
                    controls[id].frame.Size = U2(v[5], v[6], v[7], v[8])
                end
            end
        end
    end
    
    local function saveLayout()
        if type(writefile) ~= "function" then return end
        local st = {}
        for id, item in pairs(controls) do
            local p, s = item.frame.Position, item.frame.Size
            st[id] = {p.X.Scale, p.X.Offset, p.Y.Scale, p.Y.Offset, s.X.Scale, s.X.Offset, s.Y.Scale, s.Y.Offset}
        end
        pcall(function() writefile(lFile, Http:JSONEncode(st)) end)
    end
    loadLayout()
    
    local defaultControls
    pcall(function()
        local pm = require(LP:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
        defaultControls = pm:GetControls()
        defaultControls:Disable()
    end)
    
    local function updateStick(sp)
        local center = stick.AbsolutePosition + stick.AbsoluteSize/2
        local offset = Vector2.new(sp.X - center.X, sp.Y - center.Y)
        local range = math.max(20, math.min(stick.AbsoluteSize.X, stick.AbsoluteSize.Y)*0.34)
        if offset.Magnitude > range then offset = offset.Unit * range end
        moveValue = Vector2.new(offset.X/range, -offset.Y/range)
        knob.Position = U2(0.5, offset.X, 0.5, offset.Y)
        updateMoveKeys(moveValue.X, moveValue.Y)
    end
    
    local function stopStick()
        moveTouch, moveValue = nil, Vector2.zero
        updateMoveKeys(0,0)
        knob.Position = U2(0.5,0,0.5,0)
    end
    
    local function cameraRelativeMove()
        if not g.Parent then return end
        local char = LP.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local cam = workspace.CurrentCamera
        if not hum or not cam then return end
        
        -- Zıplama Modu
        if cyRuntime.CYBOR_JUMPING and hum then hum.Jump = true end
        
        if cyRuntime.CYBOR_SHIFTLOCK and root and hum.Health > 0 then
            hum.AutoRotate = false
            hum.CameraOffset = Vector3.new(1.75, 0, 0)
            local look = cam.CFrame.LookVector
            local flatLook = Vector3.new(look.X, 0, look.Z)
            if flatLook.Magnitude > 0.01 then root.CFrame = CFrame.new(root.Position, root.Position + flatLook.Unit) end
        else
            hum.AutoRotate = true
            hum.CameraOffset = Vector3.zero
        end
        
        if editMode then hum:Move(Vector3.zero, false) return end
        
        local look, right = cam.CFrame.LookVector, cam.CFrame.RightVector
        local fwd, side = Vector3.new(look.X, 0, look.Z), Vector3.new(right.X, 0, right.Z)
        if fwd.Magnitude < 0.001 or side.Magnitude < 0.001 then return end
        local dir = side.Unit * moveValue.X + fwd.Unit * moveValue.Y
        if dir.Magnitude > 1 then dir = dir.Unit end
        hum:Move(dir, false)
    end
    
    local rName = "CYBOR_PC_Move_"..tostring(LP.UserId)
    pcall(function() RS:UnbindFromRenderStep(rName) end)
    RS:BindToRenderStep(rName, Enum.RenderPriority.Input.Value+3, cameraRelativeMove)
    
    g.Destroying:Connect(function()
        updateMoveKeys(0,0)
        pcall(function() RS:UnbindFromRenderStep(rName) end)
        if defaultControls then pcall(function() defaultControls:Enable() end) end
    end)
    
    stick.InputBegan:Connect(function(input)
        if editMode or input.UserInputType ~= Enum.UserInputType.Touch then return end
        local center = stick.AbsolutePosition + stick.AbsoluteSize/2
        local delta = Vector2.new(input.Position.X - center.X, input.Position.Y - center.Y)
        if delta.Magnitude > math.min(stick.AbsoluteSize.X, stick.AbsoluteSize.Y)/2 or moveTouch then return end
        moveTouch = input updateStick(input.Position)
    end)
    
    local function controlAt(pt)
        for id, item in pairs(controls) do
            if inRect(pt, item.frame) then return id, item end
        end
        return nil
    end
    
    local cachedTools = {}
    local function refreshToolSlots()
        local char, bp = LP.Character, LP:FindFirstChild("Backpack")
        local existing, present = {}, {}
        if bp then for _, it in ipairs(bp:GetChildren()) do if it:IsA("Tool") then present[it] = true table.insert(existing, it) end end end
        if char then for _, it in ipairs(char:GetChildren()) do if it:IsA("Tool") then present[it] = true table.insert(existing, it) end end end
        for i=1,5 do if cachedTools[i] and not present[cachedTools[i]] then cachedTools[i] = nil end end
        for _, t in ipairs(existing) do
            local f = false for i=1,5 do if cachedTools[i] == t then f = true break end end
            if not f then for i=1,5 do if not cachedTools[i] then cachedTools[i] = t break end end end
        end
    end
    
    local function equipSlot(slot)
        sendVirtualKey(tostring(slot), true)
        task.delay(0.05, function() sendVirtualKey(tostring(slot), false) end)
        refreshToolSlots()
        local char = LP.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        local tTool = cachedTools[slot]
        if tTool then
            if tTool.Parent == char then hum:UnequipTools() else hum:EquipTool(tTool) end
        end
        for i=1,5 do
            local ci = controls["Slot"..i]
            if ci then local t = cachedTools[i] ci.frame.BackgroundColor3 = (t and t.Parent == char) and cols.active or cols.default end
        end
    end
    
    local function wheel(zIn)
        local cam = workspace.CurrentCamera if not cam then return end
        local est = (cam.CFrame.Position - cam.Focus.Position).Magnitude
        local nDist = math.clamp(est + (zIn and -3 or 3), 2, 25)
        LP.CameraMinZoomDistance = nDist LP.CameraMaxZoomDistance = nDist + 0.05
        task.delay(0.1, function() LP.CameraMinZoomDistance = 0.5 LP.CameraMaxZoomDistance = 128 end)
    end
    
    local function clickAtPoint(pt)
        local cam = workspace.CurrentCamera if not cam then return end
        local v = cam.ViewportSize
        sendVirtualClick(pt and pt.X or v.X/2, pt and pt.Y or v.Y/2)
    end
    
    local function isOtherGui(pt)
        local ok, arr = pcall(function() return CG:GetGuiObjectsAtPosition(math.floor(pt.X), math.floor(pt.Y)) end)
        if not ok then return false end
        for _, el in ipairs(arr) do
            if el:IsA("GuiObject") and el.Visible and el.Active and not el:IsDescendantOf(g) then return true end
        end
        return false
    end
    
    local pressed = {}
    local modifiers = {Shift = false, Ctrl = false}
    cyRuntime.CYBOR_PC_MODIFIERS = modifiers
    
    cyTrack(UIS.InputBegan:Connect(function(input, processed)
        if not g.Parent or input.UserInputType ~= Enum.UserInputType.Touch then return end
        local name, item = controlAt(input.Position)
        if editMode then
            if name then
                editTouch, editControl = input, item.frame
                editAction = inRect(input.Position, item.resize) and "resize" or "move"
                editStartPos = input.Position
                editStartValue = editAction == "move" and item.frame.Position or item.frame.Size
            end
            return
        end
        
        if name == "Joystick" then return end
        if name then
            if pressed[name] then return end
            pressed[name] = input
            if name == "Shift" then
                cyRuntime.CYBOR_SHIFTLOCK = not cyRuntime.CYBOR_SHIFTLOCK
                modifiers.Shift = cyRuntime.CYBOR_SHIFTLOCK
                reticle.Visible = cyRuntime.CYBOR_SHIFTLOCK or mouseClickMode
                item.frame.BackgroundColor3 = cyRuntime.CYBOR_SHIFTLOCK and cols.active or cols.default
                sendVirtualKey("Shift", true)
                task.delay(0.05, function() sendVirtualKey("Shift", false) end)
            elseif name == "Jump" then
                cyRuntime.CYBOR_JUMPING = true
                item.frame.BackgroundColor3 = cols.active
                sendVirtualKey("Space", true)
            elseif name == "Ctrl" then
                modifiers.Ctrl = true
                item.frame.BackgroundColor3 = cols.active
                sendVirtualKey("Ctrl", true)
            elseif string.sub(name,1,4) == "Slot" then
                equipSlot(tonumber(string.sub(name,5)))
            elseif name == "WheelUp" then
                wheel(true)
            elseif name == "WheelDown" then
                wheel(false)
            elseif name == "Mouse" then
                clickAtPoint(nil)
            end
            return
        end
        if mouseClickMode and not processed and not isOtherGui(input.Position) then
            pendingMouseTouches[input] = {position = input.Position, time = os.clock()}
        end
    end))
    
    cyTrack(UIS.InputChanged:Connect(function(input)
        if not g.Parent or input.UserInputType ~= Enum.UserInputType.Touch then return end
        if moveTouch == input and not editMode then updateStick(input.Position) end
        if editMode and editTouch == input and editControl then
            local diff = input.Position - editStartPos
            if editAction == "resize" then
                local o = editStartValue
                editControl.Size = U2(o.X.Scale, math.max(40, o.X.Offset+diff.X), o.Y.Scale, math.max(40, o.Y.Offset+diff.Y))
            else
                local o = editStartValue
                editControl.Position = U2(o.X.Scale, o.X.Offset+diff.X, o.Y.Scale, o.Y.Offset+diff.Y)
            end
        end
    end))
    
    cyTrack(UIS.InputEnded:Connect(function(input)
        if not g.Parent or input.UserInputType ~= Enum.UserInputType.Touch then return end
        if moveTouch == input then stopStick() end
        if editTouch == input then editTouch, editControl, editAction = nil, nil, nil end
        
        for id, act in pairs(pressed) do
            if act == input then
                pressed[id] = nil
                if id == "Jump" then
                    cyRuntime.CYBOR_JUMPING = false
                    sendVirtualKey("Space", false)
                    controls[id].frame.BackgroundColor3 = editMode and cols.edit or cols.default
                end
                if id == "Ctrl" then
                    modifiers.Ctrl = false
                    sendVirtualKey("Ctrl", false)
                    controls[id].frame.BackgroundColor3 = editMode and cols.edit or cols.default
                end
            end
        end
        
        local data = pendingMouseTouches[input]
        pendingMouseTouches[input] = nil
        if data and mouseClickMode and not editMode and (input.Position - data.position).Magnitude < 15 and os.clock() - data.time < 0.35 then
            if cyRuntime.CYBOR_SHIFTLOCK then clickAtPoint(nil) else clickAtPoint(input.Position) end
        end
    end))
    
    sTab:Section({Title = "PC Mode Settings"})
    sTab:Toggle({Title = "Touch Screen to Mouse Click", Default = true, Callback = function(v) mouseClickMode = v reticle.Visible = v or cyRuntime.CYBOR_SHIFTLOCK end})
    sTab:Button({Title = "Edit PC Controls", Icon = "move", Callback = function() setEditing(not editMode) end})
    sTab:Button({Title = "Save PC Layout", Icon = "save", Callback = function() saveLayout() setEditing(false) end})
    sTab:Button({Title = "Reset PC Layout", Icon = "rotate-ccw", Callback = function() restoreDefaults() setEditing(false) end})

    Tr:Button({Title = "[OP] Fling Sheriff", Icon = "shield", Callback = function() for _, p in pairs(P:GetPlayers()) do if p ~= LP and gR(p) == "S" then FP(p) break end end end})
    Tr:Button({Title = "[OP] Fling All", Icon = "users", Callback = function() task.spawn(function() for _, p in pairs(P:GetPlayers()) do if p ~= LP and p.Character then FP(p) task.wait(0.5) end end end) end})
    
    Es:Toggle({Title = "ESP Murderer", Callback = function(v) getgenv().EM = v end})
    Es:Toggle({Title = "ESP Sheriff", Callback = function(v) getgenv().ES = v end})
    Es:Toggle({Title = "ESP Innocent", Callback = function(v) getgenv().EI = v end})
    Es:Toggle({Title = "ESP Gun (Map)", Callback = function(v) getgenv().EG = v end})
    
    task.spawn(function()
        local hadGunESP, lastGunScan = false, 0
        while task.wait(0.3) and cyActive() do
            local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            for _, p in pairs(P:GetPlayers()) do
                if p ~= LP then
                    local c = p.Character
                    if c and c:FindFirstChild("HumanoidRootPart") and c:FindFirstChildOfClass("Humanoid") and c:FindFirstChildOfClass("Humanoid").Health > 0 then
                        local r = gR(p)
                        local show = (r == "M" and getgenv().EM) or (r == "S" and getgenv().ES) or (r == "I" and getgenv().EI)
                        local h, bb = c:FindFirstChild("CyESP_H"), c:FindFirstChild("CyESP_BB")
                        
                        if show then
                            if not h then
                                h = IN("Highlight", c) h.Name = "CyESP_H"
                                bb = IN("BillboardGui", c) bb.Name, bb.Size, bb.StudsOffset, bb.AlwaysOnTop = "CyESP_BB", U2(0, 200, 0, 40), Vector3.new(0, 4, 0), true
                                local l = IN("TextLabel", bb) l.Name, l.Size, l.BackgroundTransparency, l.Font, l.TextScaled, l.TextStrokeTransparency = "Txt", U2(1, 0, 1, 0), 1, Enum.Font.GothamBlack, true, 0
                            end
                            local col = r == "M" and C3(255, 20, 20) or r == "S" and C3(20, 150, 255) or C3(50, 255, 50)
                            local rt = r == "M" and "MURDERER" or r == "S" and "SHERIFF/HERO" or "INNOCENT"
                            local dt = myRoot and math.floor((myRoot.Position - c.HumanoidRootPart.Position).Magnitude) or 0
                            h.FillColor, h.OutlineColor = col, col
                            if bb and bb:FindFirstChild("Txt") then
                                bb.Txt.TextColor3 = col
                                bb.Txt.Text = string.format("[%s]\n%s [%dm]", rt, p.Name, dt)
                            end
                        else
                            if h then h:Destroy() end
                            if bb then bb:Destroy() end
                        end
                    end
                end
            end
            
            if getgenv().EG and os.clock() - lastGunScan >= 1.5 then
                lastGunScan = os.clock() hadGunESP = true
                for _, v in pairs(workspace:GetDescendants()) do
                    if v.Name == "GunDrop" and v:IsA("BasePart") and not v:FindFirstChild("CyESP_G") then
                        local bb = IN("BillboardGui", v) bb.Name, bb.Size, bb.AlwaysOnTop = "CyESP_G", U2(0, 150, 0, 30), true
                        local l = IN("TextLabel", bb) l.Size, l.BackgroundTransparency, l.TextColor3, l.Font, l.TextScaled, l.Text = U2(1, 0, 1, 0), 1, C3(255, 215, 0), Enum.Font.GothamBlack, true, "[GUN HERE]"
                    end
                end
            elseif not getgenv().EG and hadGunESP then
                hadGunESP = false
                for _, v in pairs(workspace:GetDescendants()) do
                    if v.Name == "GunDrop" and v:FindFirstChild("CyESP_G") then v.CyESP_G:Destroy() end
                end
            end
        end
    end)
end
